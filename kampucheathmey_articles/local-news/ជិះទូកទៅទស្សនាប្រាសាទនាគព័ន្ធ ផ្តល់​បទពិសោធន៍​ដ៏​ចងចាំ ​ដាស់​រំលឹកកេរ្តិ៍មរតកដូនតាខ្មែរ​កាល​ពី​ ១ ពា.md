# ជិះទូកទៅទស្សនាប្រាសាទនាគព័ន្ធ ផ្តល់​បទពិសោធន៍​ដ៏​ចងចាំ ​ដាស់​រំលឹកកេរ្តិ៍មរតកដូនតាខ្មែរ​កាល​ពី​ ១ ពាន់ឆ្នាំមុន

កាលពីសម័យបុរាណ ប្រាសាទនាគព័ន្ធគឺជាមន្ទីរពេទ្យសម្រាប់ព្យាបាលជំងឺ។ កាលនោះគ្មានផ្លូវឆ្លងដើរដូចសព្វថ្ងៃទេ មានតែមធ្យោបាយតាមទូក។ នេះបើតាមលោក យី យួក អនុប្រធានសមាគមទូកទេសចរណ៍នៅប្រាសាទនាគព័ន្ធ។

ការជិះទូកផ្តល់បទពិសោធន៍ក្នុងការមើលទេសភាពព្រៃលិចទឹក និងសត្វស្លាប ព្រមទាំងអាចឱ្យភ្ញៀវស្រមៃយល់ដឹងពីប្រវត្តិសាស្ត្រកាលពីសម័យដើម។

ពាក់ព័ន្ធបទពិសោធន៍ និងអត្ថន័យប្រវត្តិសាស្ត្រនៃប្រាសាទនាគព័ន្ធ លោក យី យួក បញ្ជាក់ថា កាលពីសម័យបុរាណ ប្រាសាទនាគព័ន្ធគឺជាមន្ទីរពេទ្យសម្រាប់ព្យាបាលជំងឺ។ កាលនោះគ្មានផ្លូវដើរដូចសព្វថ្ងៃទេ ដូច្នេះអ្នកមកព្យាបាលជំងឺទាំងអស់ត្រូវធ្វើដំណើរតាមទូក។

សហគមន៍ទូកទេសចរណ៍នៅប្រាសាទនាគព័ន្ធបង្កើតឡើងដំបូងនៅក្នុងឆ្នាំ ២០១៧ ដោយមានការសហការចូលរួមពីអាជ្ញាធរជាតិអប្សរា រដ្ឋាភិបាលអូស្ត្រាលី និងអង្គការយូណេស្កូ (UNESCO)។

បច្ចុប្បន្នសមាជិកមានចំនួនជាង ២០ នាក់ ជាប្រជាពលរដ្ឋមកពីភូមិប្លង់ និងភូមិលាងដៃ។

លោក យី យួក អនុប្រធានសមាគមទូកទេសចរណ៍នៅប្រាសាទនាគព័ន្ធ

លោក យី យួក ឲ្យដឹងថា សហគមន៍មានទូកចំនួន ៧ ដែល​ផ្តល់ជូនខ្សែទស្សនាចំនួន ៣ ជម្រើស​សម្រាប់ភ្ញៀវទេសចរជិះកម្សាន្ត។ ខ្សែទស្សនាទី១ មានរយៈពេលកន្លះម៉ោង សម្រាប់ភ្ញៀវជិះមើលទេសភាពជុំវិញ ដែលមានតម្លៃ ១២ ដុល្លារ សម្រាប់ភ្ញៀវជាតិ និង ២០ ដុល្លារ សម្រាប់ភ្ញៀវអន្តរជាតិ។

ខ្សែទស្សនាទី២ និងទី៣ មានរយៈពេល ១ ម៉ោងកន្លះ ទៅ ២ ម៉ោង ទាំងទៅទាំងមក ដែលធ្វើដំណើរទៅដល់ប្រាសាទនាគព័ន្ធ ដែលតម្លៃគឺ ២៣ ដុល្លារ សម្រាប់ភ្ញៀវជាតិ និង ៤០ ដុល្លារ សម្រាប់ភ្ញៀវអន្តរជាតិ។

លោក យី យួក ថា ការជិះទូកនេះផ្តល់បទពិសោធន៍ថ្មីប្លែកក្នុងការមើលទេសភាព ព្រៃលិចទឹក និងសត្វស្លាប ព្រមទាំងអាចឱ្យភ្ញៀវស្រមៃយល់ដឹងពីប្រវត្តិសាស្ត្រកាលពីសម័យដើម។

លោកប្រាប់ពេលវេលាល្អបំផុតសម្រាប់ការជិះទូក គឺនៅពេលព្រឹកព្រលឹម (មើលថ្ងៃរះ) និងពេលល្ងាច (មើលថ្ងៃលិច) ដែលជាពេលទេសភាពបារាយណ៍មានសម្រស់ស្អាតបំផុត។

ភ្ញៀវជាតិភាគច្រើនមកកម្សាន្តនៅថ្ងៃសៅរ៍ ថ្ងៃអាទិត្យ និងថ្ងៃបុណ្យ ចំណែកថ្ងៃធម្មតាមានតិចតួច។ សម្រាប់ភ្ញៀវអន្តរជាតិមានខ្លះៗមកតាមរយៈក្រុមហ៊ុនទេសចរណ៍។ នេះបើតាមលោក យី យួក។

ដើម្បីទាក់ទាញទេសចរនិងផ្តល់ព័ត៌មានពីសេវាកម្មនៅប្រាសាទនាគព័ន្ធ សហគមន៍បានធ្វើការផ្សព្វផ្សាយតាមរយៈការចែកខិតប័ណ្ណព័ត៌មាននៅតាមសណ្ឋាគារ និងការបង្ហោះនៅលើ TikTok និង Facebook ជាដើម។

បើតាមសមាគមទូកទេសចរណ៍នៅប្រាសាទនាគព័ន្ធ វិស័យទេសចរណ៍ជាប្រភពចំណូលសម្រាប់សមាគម ដោយសារពួកគាត់គ្មានប្រាក់ខែប្រចាំនោះទេ គឺរំពឹងតែលើចំណូលពីការរត់ទូកនេះឯង។

ខណៈដែលបច្ចុប្បន្នចំនួនភ្ញៀវនិងចំណូលមានការធ្លាក់ចុះជាងមុន ទូកទេសចរណ៍នៅប្រាសាទនាគព័ន្ធអំពាវនាវដល់ភ្ញៀវជាតិ និងអន្តរជាតិទាំងអស់ឱ្យអញ្ជើញមកកម្សាន្ត និងជិះទូករបស់សហគមន៍ ដើម្បី​ជួយទ្រទ្រង់ជីវភាពក្រុមការងារសហគមន៍ ដែលរស់នៅពឹងអាស្រ័យលើវិស័យទេសចរណ៍៕

អាជ្ញាធរជាតិអប្សរា ជួសជុលបង្កាន់ដៃនាគជ្រុងខាងលិចប្រវែង២០ម៉ែត្រ

មន្ត្រីជាន់ខ្ពស់ក្រសួងមហាផ្ទៃ ណែនាំ មន្ត្រីពាក់ព័ន្ធ និង សមត្ថកិច្ច បន្តពន្លឿនការផ្ដល់អត្តសញ្ញាណបណ្ណសញ្ជាតិខ្មែរជូនប្រជាពលរដ្ឋឲ្យបានលឿន

រដ្ឋបាលខេត្តសៀមរាប ដង្ហែក្បួនអបអរសាទរជ័យលាភីអង្គភាពផ្ដល់សេវាសាធារណៈគំរូ ក្នុងវិស័យអប់រំ និង សុខាភិបាល ឆ្នាំ ២០២៥

រដ្ឋបាលខេត្តសៀមរាប បន្តដំឡេីងភ្លេីងសញ្ញា និងស្លាកសញ្ញាចរាចរណ៍ ចំណុចដែលមានគ្រោះថ្នាក់ញឹកញាប់​
